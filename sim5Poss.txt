void * processGen(void * threadForProcess){
    do{
        if (rPaused == 0){
        Process * process;
            if (maxNumOfProc >> 0){
                maxNumOfProc --;
                pthread_mutex_lock(&poolMutex);
                i = getPID();
                if (i==-1){
                    rPaused = 1;
                }
                else{

                    process = generateProcess(i);
                    pidTable[i] = process;
                }
                pthread_mutex_unlock(&poolMutex);
                numOfProcesses --;
	            printf("GENERATOR - CREATED: [PID = %d, Priority = %d, InitialBurstTime = %d, RemainingBurstTime = %d]\n"\
    	        ,process->iPID,process->iPriority,process->iBurstTime,process->iRemainingBurstTime);		//prints out details

                rSize ++;
            }
            else 
            {   
                rPaused = 1;
                sPaused = 0;

            }

        }
    
    
    }while (numOfProcesses-1 >= 0);
    printf("GENERATOR FINISHED\n");
    gDone = 3;

void * processSim(void * threadForProcess){
    
    Process * process;
    void * data = "";
    do{
    if (sPaused == 0){
        data = NULL;
        pthread_mutex_lock(&queueMutex);
        data = removeFirst(&rList); //gets the element at the front of list
        pthread_mutex_unlock(&queueMutex);
        if (data == NULL){
            sPaused = 1;
            rPaused = 4;
        }
        else{  
            process = data; //gets the data from the element
            rSize --;
            printf("QUEUE - Removed: [Queue = READY, Size = %d, PID = %d, Priority = %d]\n",rSize,process->iPID,process->iPriority);
            if(process->iPID < numOfPriorityLevels/2){
                runNonPreemptiveProcess(process,false);
            }
            else{
                runPreemptiveProcess(process,false);
            }
            printf("SIMULATOR - CPU 0: RR [PID = %d, Priority = %d, InitialBurstTime = %d, RemainingBurstTime = %d]\n",process->iPID,process->iPriority,process->iBurstTime,process->iRemainingBurstTime);  
        
        if (data == NULL){
        }
        else if (process->iState == 4){
            tSize ++;
            int a = getDifferenceInMilliSeconds(process->oTimeCreated,process->oFirstTimeRunning);
            int b = getDifferenceInMilliSeconds(process->oTimeCreated,process->oLastTimeRunning);
            printf("SIMULATOR - CPU 0 - TERMINATED: [PID = %d, ResponseTime = %d, TurnAroundTime = %d]\n",process->iPID,a,b);
            printf("QUEUE - ADDED: [Queue = TERMINATED, Size = %d, PID = %d, Priority = %d]\n",tSize,process->iPID,process->iPriority);

            pthread_mutex_lock(&queueMutex);
            addLast(process, &tList);  //if terminated add to finished list
            pthread_mutex_unlock(&queueMutex);
            tPaused = 0;
            rPaused = 0;
            maxNumOfProc ++;
        }
        else{
            rSize ++;
            pthread_mutex_lock(&queueMutex);
            addLast(process,&rList);  //if ready then add back to the list
            pthread_mutex_unlock(&queueMutex);
            printf("QUEUE - ADDED: [Queue = READY, Size = %d, PID = %d, Priority = %d]\n",rSize,process->iPID,process->iPriority);
            }  
    }
    }
    }while (gDone != 3 || data != NULL);
    

    rPaused = 4;
    printf("SIMULATOR: Finished \n");
    tPaused = 0;
    return threadForProcess;
}

void * processTerm(void * threadForProcess){

    Process * process;
    void * data = "";
    long int avgRT = 0;
    long int avgTT = 0;
    int count = 0;
    int times = 0;
    do{
        if (tPaused == 0){
            data = NULL;
            pthread_mutex_lock(&queueMutex);
            data = removeFirst(&tList); 
            pthread_mutex_unlock(&queueMutex);
            if (data == NULL){
                tPaused = 1;
            }
            else{
                process = data;  //gets the element at the front of list
                count ++;
                tSize --;

                //errors are coming from here
                
                avgRT += getDifferenceInMilliSeconds(process->oTimeCreated,process->oFirstTimeRunning);
                avgTT += getDifferenceInMilliSeconds(process->oTimeCreated,process->oLastTimeRunning);

                printf("QUEUE - REMOVED: [Queue = TERMINATED, Size = %d, PID = %d, Priority = %d]\n",tSize,process->iPID,process->iPriority);
                printf("TERMINATION DAEMON - CLEARED: [#iTerminated = %d, PID = %d, Priority = %d]\n",count,process->iPID,process->iPriority);
                
                //int j = process->iPID;
                pthread_mutex_lock(&poolMutex);
                times += 1;
                returnPID(process->iPID);
                //destroyProcess(pidTable[j]);
                //pidTable[j] = NULL;
                pthread_mutex_unlock(&poolMutex);

                destroyProcess(process);

            }  
        }
    }while (rPaused != 4 || data != NULL);
    avgRT = avgRT / (long int)NUMBER_OF_PROCESSES;
    avgTT = avgTT / (long int)NUMBER_OF_PROCESSES;
    printf("tlist %p   rlist %p\n ",removeFirst(&tList), removeFirst(&rList));
    printf("TERMINATION DAEMON: Finished\nTERMINATION DAEMON: [Average Response Time = %ld, Average Turn Around Time = %ld]\n",avgRT,avgTT);
    return threadForProcess;
}

void pidPoolCreate(){
    for (int k = 0;k < MAX_CONCURRENT_PROCESSES;k++){
        pidPool[k] = false;    
    }
}

int getPID(){
    for (int k = 0;k < MAX_CONCURRENT_PROCESSES;k++){
        if (pidPool[k] == false){
            pidPool[k] = true; //being used
            return k;
        }
    }
    return -1;
}

void returnPID(int num){
    pidPool[num] = false;
}

}

void * processTerm(void * threadForProcess){

    Process * process;
    void * data = "";
    long int avgRT = 0;
    long int avgTT = 0;
    int count = 0;
    int times = 0;
    do{
        if (tPaused == 0){
            data = NULL;
            pthread_mutex_lock(&queueMutex);
            data = removeFirst(&tList); 
            pthread_mutex_unlock(&queueMutex);
            if (data == NULL){
                tPaused = 1;
            }
            else{
                process = data;  //gets the element at the front of list
                count ++;
                tSize --;

                //errors are coming from here
                
                avgRT += getDifferenceInMilliSeconds(process->oTimeCreated,process->oFirstTimeRunning);
                avgTT += getDifferenceInMilliSeconds(process->oTimeCreated,process->oLastTimeRunning);

                printf("QUEUE - REMOVED: [Queue = TERMINATED, Size = %d, PID = %d, Priority = %d]\n",tSize,process->iPID,process->iPriority);
                printf("TERMINATION DAEMON - CLEARED: [#iTerminated = %d, PID = %d, Priority = %d]\n",count,process->iPID,process->iPriority);
                
                //int j = process->iPID;
                pthread_mutex_lock(&poolMutex);
                times += 1;
                returnPID(process->iPID);
                //destroyProcess(pidTable[j]);
                //pidTable[j] = NULL;
                pthread_mutex_unlock(&poolMutex);

                destroyProcess(process);

            }  
        }
    }while (rPaused != 4 || data != NULL);
    avgRT = avgRT / (long int)NUMBER_OF_PROCESSES;
    avgTT = avgTT / (long int)NUMBER_OF_PROCESSES;
    printf("tlist %p   rlist %p\n ",removeFirst(&tList), removeFirst(&rList));
    printf("TERMINATION DAEMON: Finished\nTERMINATION DAEMON: [Average Response Time = %ld, Average Turn Around Time = %ld]\n",avgRT,avgTT);
    return threadForProcess;
}
