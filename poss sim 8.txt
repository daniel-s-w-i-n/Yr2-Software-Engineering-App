int main(){
    int threadId[NUMBER_OF_CPUS];
    void * status;

    pidPoolCreate();
    pthread_create((&processGenerator),NULL,processGen,NULL);
    for(int j = 0; j< NUMBER_OF_CPUS ;j++)
    {
        threadId[j] = j;
        pthread_create(&processSimulator[j],NULL,processSim,(void*)(threadId + j));
    }
    pthread_create((&boosterDaemon),NULL,boosterDae,NULL);
    pthread_create((&boosterIO),NULL,boosterIOt,NULL);
    pthread_create((&processTerminator),NULL,processTerm,NULL);
    
    pthread_join(processGenerator, &status); 
    pthread_join(processTerminator, &status); 
    pthread_join(boosterDaemon, &status); 
    pthread_join(boosterIO, &status); 
    for(int l = 0; l< NUMBER_OF_CPUS ;l++)
    {
        pthread_join(processSimulator[l], &status); 
    }
    pthread_exit(NULL);
    return 0;
}

void pidPoolCreate(){
    for (int k = 0;k < MAX_CONCURRENT_PROCESSES;k++){
        pidPool[k] = false;    
    }
}

int getPID(){
    for (int k = 0;k < MAX_CONCURRENT_PROCESSES;k++){
        if (pidPool[k] == false){
            pidPool[k] = true; 
            return k;
        }
    }
    return -1;
}

void returnPID(int num){
    pidPool[num] = false;
}

int countSize(LinkedList list){
    Element * b = getHead(list);
    int count = 0;
    while(b != NULL){
        count ++;
        b = getNext(b);
    }
    return count;
}

void * boosterIOt(void * threadForProcess){
    Process * process;
    while(rPaused != 4){
        if (sPaused == 0){
            usleep(ioDelay);
            int amountInQueue = 0;
            //get the processes from the device queue
            pthread_mutex_lock(&IOMutex);
            pthread_mutex_lock(&queueMutex);
            for (int i = 0; i<numOfDev ; i++){
                amountInQueue = countSize((IOQueues[i]));
                if (amountInQueue > 0){
                    process = removeFirst(&(IOQueues[i]));
                     //put the process in the queue
                     printf("I/O DAEMON - UNBLOCKED: [PID = %d, Priority = %d]\n",process->iPID,process->iPriority);
                    
                    addFirst(process, &(priorityQueues[process->iPriority]));
                    
                }
            }
            pthread_mutex_unlock(&queueMutex);
            pthread_mutex_unlock(&IOMutex);
    
    }}
    printf("I/O DAEMON: Finished\n");
    return threadForProcess;
}

void * boosterDae(void * threadForProcess){
    int amountInQueue;
    void * data;
    Process * process;
    while(rPaused != 4)
    {
        if (sPaused == 0){
        usleep(boostDelay);
        amountInQueue = 0;
        pthread_mutex_lock(&queueMutex);
        for (int l = 0 ; l < numOfPriorityLevels/2; l++){
            amountInQueue = countSize((priorityQueues[numOfPriorityLevels-l-1]));
           
            if(amountInQueue > 0){
                
                data = removeFirst(&(priorityQueues[numOfPriorityLevels-l-1]));
                process = data;
                printf("QUEUE - REMOVED: [Queue = READY %d, Size = %d, PID = %d, Priority = %d]\n"
                ,process->iPID,amountInQueue,process->iPID,process->iPID);
                printf("BOOSTER DAEMON: [PID = %d, Priority = %d, InitialBurstTime = %d, RemainingBurstTime = %d] => Boosted to Level 8\n"
                ,process->iPID,process->iPriority,process->iBurstTime,process->iRemainingBurstTime);
                printf("QUEUE - ADDED: [Queue = READY 8, Size = %d, PID = %d, Priority = %d]\n"
                ,amountInQueue,process->iPID,process->iPID);
                
                addFirst(process, &(priorityQueues[7]));
            }
        
        }
        pthread_mutex_unlock(&queueMutex);
        }
    }
    printf("BOOSTER DAEMON: Finished\n");
    return threadForProcess;
}


void * processGen(void * threadForProcess){
    printf("BOOSTER DAEMON: Created\n");
    do{
        int queueSize = 0;
        if (rPaused == 0){
        Process * process;
            if (maxNumOfProc >> 0){
                maxNumOfProc --;
                pthread_mutex_lock(&poolMutex);
                i = getPID();
                pthread_mutex_unlock(&poolMutex);
                if (i==-1){
                    rPaused = 1;
                }
                else{

                    process = generateProcess(i);

                    pthread_mutex_unlock(&poolMutex);

	            printf("GENERATOR - CREATED: [PID = %d, Priority = %d, InitialBurstTime = %d, RemainingBurstTime = %d]\n"\
    	        ,process->iPID,process->iPriority,process->iBurstTime,process->iRemainingBurstTime);

                {
                        printf("invalid Process\n\n");
                        maxNumOfProc ++;
                }
                else{

                    pthread_mutex_lock(&queueMutex);
                    addLast(process, &(priorityQueues[process->iPriority]));
                    pthread_mutex_unlock(&queueMutex);

                    queueSize = countSize((priorityQueues[process->iPriority]));

                    printf("GENERATOR - ADDED TO TABLE: [PID = %d, Priority = %d, Initial BurstTime = %d, Remaining BurstTime = %d]\n"\

                    ,process->iPID,process->iPriority,process->iBurstTime,process->iRemainingBurstTime);


                
                    rSize++;
                }
                }
            }
            else

                rPaused = 1;

        }
    
    
    }while (numOfProcesses >> 0);

}




void * processSim(/*void * threadForProcess,*/void * arg){
    
    Process * process;
    void * data = "";
    int queueSize;
    int queueUsed = 0;
    int * id = (int *) arg;

    do{
    //if (sPaused == 0){
        data = NULL;
        pthread_mutex_lock(&queueMutex);
        for(int l = 0 ;l<numOfPriorityLevels ; l++){
            data = removeFirst(&(priorityQueues[l]));
            if (data != NULL){
                queueUsed = l;
                break;
            }
        }
        pthread_mutex_unlock(&queueMutex);
        if (data == NULL){
            sPaused = 1;
            rPaused = 0;
            tPaused = 0;
        }
        else{  
            process = data; //gets the data from the element
            rSize --;
            queueSize = countSize((priorityQueues[process->iPriority]));
            printf("QUEUE - REMOVED: [Queue = READY %d, Size = %d, PID = %d, Priority = %d]\n",process->iPriority,queueSize,process->iPID,process->iPriority);
            if(process->iPriority < numOfPriorityLevels/2){
                runNonPreemptiveProcess(process,true);
                printf("SIMULATOR - CPU %d: FCFS [PID = %d, Priority = %d, InitialBurstTime = %d, RemainingBurstTime = %d]\n",* id,process->iPID,process->iPriority,process->iBurstTime,process->iRemainingBurstTime);  
            }
            else{
                runPreemptiveProcess(process,true);
                printf("SIMULATOR - CPU %d: RR [PID = %d, Priority = %d, InitialBurstTime = %d, RemainingBurstTime = %d]\n",* id,process->iPID,process->iPriority,process->iBurstTime,process->iRemainingBurstTime);  
            }
            
            if (data == NULL){
            }
            else if (process->iState == 4){
                tSize ++;
                int a = getDifferenceInMilliSeconds(process->oTimeCreated,process->oFirstTimeRunning);
                int b = getDifferenceInMilliSeconds(process->oTimeCreated,process->oLastTimeRunning);
                printf("SIMULATOR - CPU %d - TERMINATED: [PID = %d, ResponseTime = %d, TurnAroundTime = %d]\n",* id,process->iPID,a,b);
                printf("QUEUE - ADDED: [Queue = TERMINATED, Size = %d, PID = %d, Priority = %d]\n",tSize,process->iPID,process->iPriority);

                pthread_mutex_lock(&queueMutex);
                addLast(process, &tList);  //if terminated add to finished list
                pthread_mutex_unlock(&queueMutex);
                tPaused = 0;
                rPaused = 0;
                maxNumOfProc ++;
                }
            else if (process->iState == 3){
                    pthread_mutex_lock(&IOMutex);
                    addLast(process,&(IOQueues[process->iDeviceID]));
                    pthread_mutex_unlock(&IOMutex);
                }
            else{
                rSize ++;
                pthread_mutex_lock(&queueMutex);
                addLast(process,&(priorityQueues[queueUsed]));  //if ready then add back to the list
                pthread_mutex_unlock(&queueMutex);
                queueSize = countSize((priorityQueues[process->iPriority]));
                printf("QUEUE - ADDED: [Queue = READY %d, Size = %d, PID = %d, Priority = %d]\n",process->iPriority,queueSize,process->iPID,process->iPriority);
                }  
            }
        //}
        tPaused = 0;
    }while (gDone != 3 || data != NULL || c < NUMBER_OF_PROCESSES);
    

    rPaused = 4;
    printf("SIMULATOR: Finished \n");
    tPaused = 0;
    return 0;//threadForProcess;
}

void * processTerm(void * threadForProcess){

    Process * process;
    void * data = "";
    double avgRT = 0;
    double avgTT = 0;
    int count = 0;
    int times = 0;
    do{
        //if (tPaused == 0){
            data = NULL;
            pthread_mutex_lock(&queueMutex);
            data = removeFirst(&tList); 
            pthread_mutex_unlock(&queueMutex);
            if (data == NULL){
                tPaused = 1;
                sPaused = 0;
            }
            else{
                process = data;  //gets the element at the front of list
                count ++;
                tSize --;
                c++;
                
                avgRT += getDifferenceInMilliSeconds(process->oTimeCreated,process->oFirstTimeRunning);
                avgTT += getDifferenceInMilliSeconds(process->oTimeCreated,process->oLastTimeRunning);

                printf("QUEUE - REMOVED: [Queue = TERMINATED, Size = %d, PID = %d, Priority = %d]\n",tSize,process->iPID,process->iPriority);
                printf("TERMINATION DAEMON - CLEARED: [#iTerminated = %d, PID = %d, Priority = %d]\n",count,process->iPID,process->iPriority);
                
                pthread_mutex_lock(&poolMutex);
                times += 1;
                int pid = process->iPID;
                returnPID(pid);
                
                if (process!= NULL){
                    destroyProcess(pidTable[pid]);
                }
                
                pthread_mutex_unlock(&poolMutex);
                

            }  
        //}
    }while (rPaused != 4 || data != NULL);
    
    avgRT = avgRT / NUMBER_OF_PROCESSES;
    avgTT = avgTT / NUMBER_OF_PROCESSES;
    printf("TERMINATION DAEMON: Finished\nTERMINATION DAEMON: [Average Response Time = %f, Average Turn Around Time = %f]\n",avgRT,avgTT);
    tPaused = 5;
    return threadForProcess;
}
